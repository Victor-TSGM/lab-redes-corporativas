// Criando porta trunk com CORE e VLAN permitidas e native vlan.

interface range gi0/0 - 1
 description UPLINK-CORE
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,99
 no shutdown

------------------------------------------------------------------------------
// Criando portas de acesso para dispositivos finais

interface gi1/0
 description PC1
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast edge                // portfast destinado para dispositivos finais
 spanning-tree bpduguard enable             // se ligado um outro switch nessa porta ela é automaticamente desativada
 no shutdown
!
interface gi1/1
 description PC2
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast edge                // portfast destinado para dispositivos finais
 spanning-tree bpduguard enable             // se ligado um outro switch nessa porta ela é automaticamente desativada
 no shutdown

-----------------------------------------------------------------------------------
// Desabilitando portas não utilizadas e inserindo em VLAN vazia para segmentar da rede principal

interface range gi1/2 - 3
 description NAO-USADA
 switchport mode access
 switchport access vlan 999
 shutdown


-----------------------------------------------------------------------------------
// Configurar switch na VLAN 99 para gerenciamento

interface vlan 99
 ip address 10.10.99.12 255.255.255.0
 no shutdown
!
ip default-gateway 10.10.99.1
